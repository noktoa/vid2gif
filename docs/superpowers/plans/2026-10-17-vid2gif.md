# vid2gif Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Go CLI that downloads a segment of an online video with yt-dlp and encodes it to GIF, animated WebP, MP4 and/or MOV with ffmpeg, using pinned, self-installed copies of both tools.

**Architecture:** A single `package main` binary. Pure functions (timestamp parsing, ffmpeg argument building, naming, output planning, version matching) carry the logic and are unit-tested. A `pipeline` struct holds the three side-effecting steps (title lookup, download, encode) as function fields, so the orchestration can be tested with fakes and run for real with yt-dlp/ffmpeg. `deps.json` pins tool versions per platform; `deps install` downloads, hash-checks and extracts them into `bin/`.

**Tech Stack:** Go (standard library) plus `github.com/ulikunitz/xz` (the only non-stdlib dependency, for the Linux ffmpeg `.tar.xz`); yt-dlp and ffmpeg as external tools; Docker for Linux integration tests.

**Spec:** `docs/superpowers/specs/2026-10-17-vid2gif-design.md`

## Global Constraints

- Platforms: `windows/amd64`, `linux/amd64`, `linux/arm64` only. Other platforms: `deps install` exits 1 with "unsupported platform; use `--no-pin` with your own yt-dlp and ffmpeg on PATH".
- Dependencies: Go standard library plus `github.com/ulikunitz/xz` only. No cobra, no other modules.
- Exit codes: `0` success (including skipped outputs), `1` runtime failure, `2` usage error.
- Formats: comma-separated subset of `gif,webp,mp4,mov`; default `gif`. Default `--out` is `./gifs`; default `--quality` is `medium`; default `--end` is start + 5 seconds.
- Quality presets (fps / max width / GIF dither / WebP quality / CRF): low 10 / 320 / bayer / 50 / 28; medium 15 / 480 / bayer / 70 / 23; high 24 / 720 / sierra2_4a / 90 / 18. Width is a cap (never upscale). Presets apply to GIF and WebP only; mp4/mov keep source resolution and fps. There are no `--fps`/`--width` overrides.
- yt-dlp must always receive `--ffmpeg-location <pinned ffmpeg>` and `--` before the URL.
- Existing output files are skipped with a message (exit 0) unless `--overwrite`. If every output is skipped, nothing is downloaded.
- Pins (verbatim from the spec): yt-dlp `2026.08.19`; ffmpeg `N-127588-g0b19c65afe` (FFmpeg-Builds `autobuild-2026-10-16-18-04`). `deps.json` lives next to the executable (fallback: current directory); `bin/` is next to `deps.json`.
- Temp files go in `<out>/.vid2gif-tmp-*` and are always removed.
- ffmpeg is downloaded at install time, never redistributed with this repo.

## Review Focus

Failure modes the spec implies that are most likely to bite a user. Each has a test in the owning task.

1. **URL that starts with `-`** must not be parsed as a yt-dlp option (always passed after `--`). Tested in Task 6.
2. **`--name` containing `/` or `\`** must be rejected as a usage error rather than writing outside `--out`. Tested in Task 7.
3. **Titles that are empty, all punctuation, very long, or contain Windows-invalid characters** must produce a valid file name. Tested in Task 5.
4. **Source video with odd dimensions or no audio track** must still encode to mp4/mov (even-dimension scale, `-c:a aac` with no audio stream). Tested in Task 8 (real ffmpeg).
5. **A failed download or encode, or Ctrl+C, must not leave `.vid2gif-tmp-*` or partial output files behind.** Tested in Task 6.

package main

import (
	"fmt"
	"math"
	"strconv"
	"strings"
)

// parseTimestamp parses SS, MM:SS or HH:MM:SS into seconds. Only the last
// component may have a fractional part; minutes and seconds after the first
// component must be below 60.
func parseTimestamp(s string) (float64, error) {
	bad := fmt.Errorf("invalid timestamp %q (use SS, MM:SS or HH:MM:SS)", s)
	parts := strings.Split(strings.TrimSpace(s), ":")
	if len(parts) > 3 {
		return 0, bad
	}
	total := 0.0
	for i, p := range parts {
		if p == "" || strings.Trim(p, "0123456789.") != "" {
			return 0, bad
		}
		var v float64
		var err error
		if i == len(parts)-1 {
			v, err = strconv.ParseFloat(p, 64)
		} else {
			var n int
			n, err = strconv.Atoi(p)
			v = float64(n)
		}
		if err != nil || (i > 0 && v >= 60) {
			return 0, bad
		}
		total = total*60 + v
	}
	return total, nil
}

// fmtSeconds renders seconds compactly (millisecond precision) for file names
// and yt-dlp section arguments.
func fmtSeconds(sec float64) string {
	return strconv.FormatFloat(math.Round(sec*1000)/1000, 'f', -1, 64)
}
```

package main

import (
	"fmt"
	"strconv"
)

type preset struct {
	fps, width  int
	dither      string
	webpQuality int
	crf         int
}

var presets = map[string]preset{
	"low":    {10, 320, "bayer", 50, 28},
	"medium": {15, 480, "bayer", 70, 23},
	"high":   {24, 720, "sierra2_4a", 90, 18},
}

type encodeOpts struct {
	format string
	preset preset
}

// ffmpegArgs builds the ffmpeg argv (without the program name) to encode the
// already-trimmed clip `in` to `out`. The single quotes inside -vf are
// ffmpeg filter-graph quoting, so the commas in min() survive.
func ffmpegArgs(in, out string, o encodeOpts) []string {
	args := []string{"-hide_banner", "-loglevel", "error", "-y", "-i", in}
	switch o.format {
	case "gif", "webp":
		vf := fmt.Sprintf("fps=%d,scale=w='min(%d,iw)':h=-1:flags=lanczos", o.preset.fps, o.preset.width)
		if o.format == "gif" {
			vf += ",split[a][b];[a]palettegen[p];[b][p]paletteuse=dither=" + o.preset.dither
			args = append(args, "-an", "-vf", vf, "-loop", "0")
		} else {
			args = append(args, "-an", "-vf", vf, "-c:v", "libwebp_anim",
				"-quality", strconv.Itoa(o.preset.webpQuality), "-loop", "0")
		}
	case "mp4", "mov":
		// yuv420p needs even dimensions
		args = append(args, "-c:v", "libx264", "-crf", strconv.Itoa(o.preset.crf),
			"-preset", "medium", "-pix_fmt", "yuv420p", "-vf", "scale=trunc(iw/2)*2:trunc(ih/2)*2",
			"-c:a", "aac", "-b:a", "128k", "-movflags", "+faststart")
	}
	return append(args, out)
}
```

package main

import (
	"encoding/json"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"runtime"
	"strings"
)

type asset struct {
	URL     string `json:"url"`
	SHA256  string `json:"sha256"`
	Archive string `json:"archive"` // none | zip | tar.xz
}

type tool struct {
	Version   string           `json:"version"`
	Platforms map[string]asset `json:"platforms"`
}

type deps map[string]tool

func loadDeps(path string) (deps, error) {
	b, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var d deps
	if err := json.Unmarshal(b, &d); err != nil {
		return nil, fmt.Errorf("%s: %w", path, err)
	}
	return d, nil
}

func platformKey() string { return runtime.GOOS + "-" + runtime.GOARCH }

func (t tool) assetFor(key string) (asset, error) {
	a, ok := t.Platforms[key]
	if !ok {
		return asset{}, fmt.Errorf("unsupported platform %s; use --no-pin with your own yt-dlp and ffmpeg on PATH", key)
	}
	return a, nil
}

func exeName(name string) string {
	if runtime.GOOS == "windows" {
		return name + ".exe"
	}
	return name

func firstLine(s string) string {
	line, _, _ := strings.Cut(strings.TrimSpace(s), "\n")
	return strings.TrimSpace(line)

// versionMatches compares a tool's version output with its pin. yt-dlp must
// match exactly; ffmpeg's "ffmpeg version <build>-<date> ..." line must start
// with the pinned build.
func versionMatches(name, pin, out string) bool {
	if name == "ffmpeg" {
		return strings.HasPrefix(strings.TrimPrefix(firstLine(out), "ffmpeg version "), pin)
	}
	return strings.TrimSpace(out) == pin
}

func versionFlag(name string) string {
	if name == "ffmpeg" {
		return "-version"
	}
	return "--version"
}

func runVersion(path, name string) (string, error) {
	out, err := exec.Command(path, versionFlag(name)).Output()
	return string(out), err
}

// resolver finds a tool (bin/ next to deps.json first, then PATH) and, unless
// noPin, checks it against the pinned version.
type resolver struct {
	baseDir  string
	lookPath func(string) (string, error)
	version  func(path, name string) (string, error)
}

func newResolver(baseDir string) resolver {
	return resolver{baseDir: baseDir, lookPath: exec.LookPath, version: runVersion}
}

func (r resolver) resolve(name string, t tool, noPin bool) (string, error) {
	binDir := filepath.Join(r.baseDir, "bin")
	path := filepath.Join(binDir, exeName(name))
	if _, err := os.Stat(path); err != nil {
		p, err := r.lookPath(name)
		if err != nil {
			return "", fmt.Errorf("%s not found in %s or on PATH; run `vid2gif deps install`", name, binDir)
		}
		path = p
	}
	if noPin {
		return path, nil
	}
	out, err := r.version(path, name)
	if err != nil {
		return "", fmt.Errorf("%s: running version check: %w", path, err)
	}
	if !versionMatches(name, t.Version, out) {
		return "", fmt.Errorf("%s version mismatch: pinned %s, found %s at %s; run `vid2gif deps install`",
			name, t.Version, firstLine(out), path)
	}
	return path, nil
}

	"archive/tar"

	"encoding/hex"

// downloadVerified streams url to dest while hashing it. dest is removed on
// any failure, including a SHA-256 mismatch.
func downloadVerified(url, wantSHA, dest string) error {
	resp, err := http.Get(url)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("GET %s: %s", url, resp.Status)
	}
	f, err := os.Create(dest)
	if err != nil {
		return err
	}
	h := sha256.New()
	_, err = io.Copy(io.MultiWriter(f, h), resp.Body)
	if cerr := f.Close(); err == nil {
		err = cerr
	}
	if err != nil {
		os.Remove(dest)
		return err
	}
	if got := hex.EncodeToString(h.Sum(nil)); !strings.EqualFold(got, wantSHA) {
		os.Remove(dest)
		return fmt.Errorf("sha256 mismatch for %s: got %s, want %s", url, got, wantSHA)
	}
	return nil
}

func isBinEntry(name, wantBase string) bool {
	name = strings.ReplaceAll(name, "\\", "/")
	return path.Base(name) == wantBase && path.Base(path.Dir(name)) == "bin"
}

func writeExe(dest string, r io.Reader) error {
	f, err := os.OpenFile(dest, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0o755)
	if err != nil {
		return err
	}
	_, err = io.Copy(f, r)
	if cerr := f.Close(); err == nil {
		err = cerr
	}
	return err
}

// extractBinary copies the archive entry <anything>/bin/<wantBase> to dest.
func extractBinary(archivePath, kind, wantBase, dest string) error {
	notFound := fmt.Errorf("%s not found in %s", wantBase, archivePath)
	switch kind {
	case "zip":
		zr, err := zip.OpenReader(archivePath)
		if err != nil {
			return err
		}
		defer zr.Close()
		for _, f := range zr.File {
			if f.FileInfo().IsDir() || !isBinEntry(f.Name, wantBase) {
				continue
			}
			rc, err := f.Open()
			if err != nil {
				return err
			}
			defer rc.Close()
			return writeExe(dest, rc)
		}
		return notFound
	case "tar.xz":
		f, err := os.Open(archivePath)
		if err != nil {
			return err
		}
		defer f.Close()
		xr, err := xz.NewReader(f)
		if err != nil {
			return err
		}
		tr := tar.NewReader(xr)
		for {
			h, err := tr.Next()
			if err == io.EOF {
				return notFound
			}
			if err != nil {
				return err
			}
			if h.Typeflag == tar.TypeReg && isBinEntry(h.Name, wantBase) {
				return writeExe(dest, tr)
			}
		}
	}
	return fmt.Errorf("unknown archive kind %q", kind)
}

// installTool makes binDir/<name> match the pin: no-op if it already does,
// otherwise download, verify, extract and re-check the version.
func installTool(name string, t tool, binDir string, log io.Writer) error {
	a, err := t.assetFor(platformKey())
	if err != nil {
		return err
	}
	exe := exeName(name)
	dest := filepath.Join(binDir, exe)
	if out, err := runVersion(dest, name); err == nil && versionMatches(name, t.Version, out) {
		fmt.Fprintf(log, "%s %s already installed\n", name, t.Version)
		return nil
	}
	if err := os.MkdirAll(binDir, 0o755); err != nil {
		return err
	}
	fmt.Fprintf(log, "installing %s %s ...\n", name, t.Version)
	if a.Archive == "none" {
		if err := downloadVerified(a.URL, a.SHA256, dest); err != nil {
			return err
		}
		if err := os.Chmod(dest, 0o755); err != nil {
			return err
		}
	} else {
		tmp := filepath.Join(binDir, "."+name+".download")
		if err := downloadVerified(a.URL, a.SHA256, tmp); err != nil {
			return err
		}
		defer os.Remove(tmp)
		if err := extractBinary(tmp, a.Archive, exe, dest); err != nil {
			return err
		}
	}
	out, err := runVersion(dest, name)
	if err != nil {
		return fmt.Errorf("%s: running version check: %w", dest, err)
	}
	if !versionMatches(name, t.Version, out) {
		return fmt.Errorf("installed %s reports %q, expected %s", name, firstLine(out), t.Version)
	}
	fmt.Fprintf(log, "installed %s\n", dest)
	return nil
}

import (
	"fmt"
	"os"
	"path/filepath"
	"slices"
	"strings"
	"unicode"
)

var allFormats = []string{"gif", "webp", "mp4", "mov"}

func parseFormats(s string) ([]string, error) {
	var out []string
	for _, f := range strings.Split(s, ",") {
		f = strings.ToLower(strings.TrimSpace(f))
		if !slices.Contains(allFormats, f) {
			return nil, fmt.Errorf("unknown format %q (choose from gif, webp, mp4, mov)", f)
		}
		if !slices.Contains(out, f) {
			out = append(out, f)
		}
	}
	return out, nil
}

var invalidFileChars = strings.NewReplacer(`<`, "_", `>`, "_", `:`, "_", `"`, "_", `/`, "_", `\`, "_", `|`, "_", `?`, "_", `*`, "_")

// sanitizeTitle makes a video title safe as a Windows/Linux file name.
func sanitizeTitle(s string) string {
	s = invalidFileChars.Replace(strings.TrimSpace(s))
	s = strings.Map(func(r rune) rune {
		if unicode.IsControl(r) {
			return '_'
		}
		return r
	}, s)
	s = strings.Join(strings.Fields(s), "_")
	if r := []rune(s); len(r) > 80 {
		s = string(r[:80])
	}
	s = strings.TrimRight(s, ". ")
	if s == "" {
		return "clip"
	}
	return s
}

func buildName(title string, start, end float64) string {
	return sanitizeTitle(title) + "_" + fmtSeconds(start) + "-" + fmtSeconds(end)
}

func outPath(outDir, name, format string) string {
	return filepath.Join(outDir, name+"."+format)
}

// planOutputs splits formats into those to write and those skipped because
// the output file already exists (unless overwrite is set).
func planOutputs(outDir, name string, formats []string, overwrite bool) (todo, skipped []string) {
	for _, f := range formats {
		if _, err := os.Stat(outPath(outDir, name, f)); err == nil && !overwrite {
			skipped = append(skipped, f)
		} else {
			todo = append(todo, f)
		}
	}
	return todo, skipped
}

package main

import (
	"context"
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"slices"
	"strings"
)

type clipOpts struct {
	URL        string
	Start, End float64
	Formats    []string
	OutDir     string
	Name       string // empty = derive from the video title
	Quality    string
	Overwrite  bool
	NoPin      bool
}

// pipeline holds the side-effecting steps as fields so run() is testable
// with fakes; newPipeline wires in the real yt-dlp and ffmpeg.
type pipeline struct {
	title    func(ctx context.Context, url string) (string, error)
	download func(ctx context.Context, o clipOpts, tmpDir string, wantAudio bool) (string, error)
	encode   func(ctx context.Context, in, out string, eo encodeOpts) error
	out      io.Writer
}

func (p pipeline) run(ctx context.Context, o clipOpts) error {
	name := o.Name
	if name == "" {
		title, err := p.title(ctx, o.URL)
		if err != nil {
			return fmt.Errorf("fetching title: %w", err)
		}
		name = buildName(title, o.Start, o.End)
	}
	todo, skipped := planOutputs(o.OutDir, name, o.Formats, o.Overwrite)
	for _, f := range skipped {
		fmt.Fprintf(p.out, "skipped %s (already exists; use --overwrite to replace)\n", outPath(o.OutDir, name, f))
	}
	if len(todo) == 0 {
		fmt.Fprintf(p.out, "wrote 0, skipped %d\n", len(skipped))
		return nil
	}
	if err := os.MkdirAll(o.OutDir, 0o755); err != nil {
		return err
	}
	tmp, err := os.MkdirTemp(o.OutDir, ".vid2gif-tmp-")
	if err != nil {
		return err
	}
	defer os.RemoveAll(tmp)

	wantAudio := slices.ContainsFunc(todo, func(f string) bool { return f == "mp4" || f == "mov" })
	clip, err := p.download(ctx, o, tmp, wantAudio)
	if err != nil {
		return err
	}

	wrote, failed := 0, 0
	for _, f := range todo {
		if err := ctx.Err(); err != nil {
			return err
		}
		final := outPath(o.OutDir, name, f)
		part := filepath.Join(tmp, name+"."+f)
		if err := p.encode(ctx, clip, part, encodeOpts{f, presets[o.Quality]}); err != nil {
			fmt.Fprintf(p.out, "failed %s: %v\n", final, err)
			failed++
			continue
		}
		if err := os.Rename(part, final); err != nil {
			fmt.Fprintf(p.out, "failed %s: %v\n", final, err)
			failed++
			continue
		}
		size := int64(0)
		if st, err := os.Stat(final); err == nil {
			size = st.Size()
		}
		fmt.Fprintf(p.out, "wrote %s (%d KB)\n", final, size/1024)
		wrote++
	}
	fmt.Fprintf(p.out, "wrote %d, skipped %d\n", wrote, len(skipped))
	if failed > 0 {
		return fmt.Errorf("%d of %d outputs failed", failed, len(todo))
	}
	return nil
}

// ytdlpTitleArgs asks yt-dlp for the video title only (no download).
func ytdlpTitleArgs(url string) []string {
	return []string{"--print", "title", "--no-playlist", "--", url}
}

// ytdlpDownloadArgs downloads just [Start, End]. --ffmpeg-location is
// mandatory so yt-dlp uses the pinned ffmpeg, and the URL follows `--` so a
// value starting with "-" is never read as an option.
func ytdlpDownloadArgs(o clipOpts, ffmpegPath, tmpDir string, wantAudio bool) []string {
	format := "bv*/b"
	if wantAudio {
		format = "bv*+ba/b"
	}
	return []string{
		"--no-playlist",
		"-f", format,
		"--download-sections", fmt.Sprintf("*%s-%s", fmtSeconds(o.Start), fmtSeconds(o.End)),
		"--force-keyframes-at-cuts",
		"--merge-output-format", "mkv",
		"--ffmpeg-location", ffmpegPath,
		"-o", filepath.Join(tmpDir, "clip.%(ext)s"),
		"--", o.URL,
	}
}

// pickClipFile finds yt-dlp's output (clip.<ext>), ignoring partial files.
func pickClipFile(dir string) (string, error) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		return "", err
	}
	for _, e := range entries {
		n := e.Name()
		if strings.HasPrefix(n, "clip.") && !strings.HasSuffix(n, ".part") && !strings.HasSuffix(n, ".ytdl") {
			return filepath.Join(dir, n), nil
		}
	}
	return "", fmt.Errorf("yt-dlp produced no clip file in %s", dir)
}

func withHint(err error, hint string) error {
	if hint == "" {
		return err
	}
	return fmt.Errorf("%w\n%s", err, hint)
}

func ffmpegEncode(ffmpeg string) func(ctx context.Context, in, out string, eo encodeOpts) error {
	return func(ctx context.Context, in, out string, eo encodeOpts) error {
		b, err := exec.CommandContext(ctx, ffmpeg, ffmpegArgs(in, out, eo)...).CombinedOutput()
		if err != nil {
			return fmt.Errorf("ffmpeg: %w\n%s", err, strings.TrimSpace(string(b)))
		}
		return nil
	}
}

// newPipeline wires in the real tools. hint is appended to yt-dlp failures
// (e.g. how to bump the pin).
func newPipeline(ytdlp, ffmpeg, hint string, out io.Writer) pipeline {
	return pipeline{
		title: func(ctx context.Context, url string) (string, error) {
			cmd := exec.CommandContext(ctx, ytdlp, ytdlpTitleArgs(url)...)
			cmd.Stderr = os.Stderr
			b, err := cmd.Output()
			if err != nil {
				return "", withHint(fmt.Errorf("yt-dlp: %w", err), hint)
			}
			return firstLine(string(b)), nil
		},
		download: func(ctx context.Context, o clipOpts, tmpDir string, wantAudio bool) (string, error) {
			cmd := exec.CommandContext(ctx, ytdlp, ytdlpDownloadArgs(o, ffmpeg, tmpDir, wantAudio)...)
			cmd.Stdout, cmd.Stderr = os.Stdout, os.Stderr
			if err := cmd.Run(); err != nil {
				return "", withHint(fmt.Errorf("yt-dlp: %w", err), hint)
			}
			return pickClipFile(tmpDir)
		},
		encode: ffmpegEncode(ffmpeg),
		out:    out,
	}
}

func (e usageError) Error() string { return string(e) }

// newClipFlags defines the clip flags. Its Usage is the full help text
// (overview, example, then every flag with its default); the flag package
// prints it on -h and on flag parse errors, and `vid2gif help` calls it too,
// so the flag list can never drift from the code.
func newClipFlags(o *clipOpts, start, end, formats *string) *flag.FlagSet {
	fs := flag.NewFlagSet("clip", flag.ContinueOnError)
	fs.SetOutput(os.Stderr)
	fs.StringVar(start, "start", "", "start time: SS, MM:SS or HH:MM:SS (required)")
	fs.StringVar(end, "end", "", "end time (default: start + 5s)")
	fs.StringVar(formats, "format", "gif", "comma-separated: gif,webp,mp4,mov")
	fs.StringVar(&o.OutDir, "out", "./gifs", "output directory (created if missing)")
	fs.StringVar(&o.Name, "name", "", "file name without extension (default <title>_<start>-<end>)")
	fs.StringVar(&o.Quality, "quality", "medium", "low, medium or high")
	fs.BoolVar(&o.Overwrite, "overwrite", false, "replace existing output files (default: skip them)")
	fs.BoolVar(&o.NoPin, "no-pin", false, "use yt-dlp and ffmpeg from PATH without version checks")
	fs.Usage = func() {
		fmt.Fprint(fs.Output(), usage, "\nFlags for clip:\n")
		fs.PrintDefaults()
	}
	return fs
}

// parseClipArgs expects `<url> [flags]`.
func parseClipArgs(args []string) (clipOpts, error) {
	if len(args) == 0 {
		return clipOpts{}, usageError("missing <url>")
	}
	var o clipOpts
	var start, end, formats string
	fs := newClipFlags(&o, &start, &end, &formats)
	rest := args
	if !strings.HasPrefix(args[0], "-") {
		o.URL, rest = args[0], args[1:]
	}
	if err := fs.Parse(rest); err != nil {
		if errors.Is(err, flag.ErrHelp) {
			return o, err
		}
		return o, usageError(err.Error())
	}
	if o.URL == "" {
		return o, usageError("the URL must come first: vid2gif clip <url> --start T")
	}
	if fs.NArg() > 0 {
		return o, usageError("unexpected argument " + fs.Arg(0))
	}
	if start == "" {
		return o, usageError("--start is required")
	}
	var err error
	if o.Start, err = parseTimestamp(start); err != nil {
		return o, usageError("--start: " + err.Error())
	}
	if end == "" {
		o.End = o.Start + 5
	} else {
		if o.End, err = parseTimestamp(end); err != nil {
			return o, usageError("--end: " + err.Error())
		}
		if o.End <= o.Start {
			return o, usageError("--end must be after --start")
		}
	}
	if o.Formats, err = parseFormats(formats); err != nil {
		return o, usageError("--format: " + err.Error())
	}
	if _, ok := presets[o.Quality]; !ok {
		return o, usageError(fmt.Sprintf("unknown --quality %q (choose low, medium or high)", o.Quality))
	}
	if strings.ContainsAny(o.Name, `/\`) {
		return o, usageError("--name must not contain path separators; use --out for the directory")
	}
	return o, nil
}

		hint = fmt.Sprintf("yt-dlp is pinned at %s. If this is a site-extraction error, bump the pin in deps.json or retry with --no-pin.", d["yt-dlp"].Version)
	}
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()
	if err := newPipeline(ytdlp, ffmpeg, hint, stdout).run(ctx, o); err != nil {
		fmt.Fprintln(stderr, "error:", err)
		return 1
	}
	return 0
}

func realMain(args []string, stdout, stderr io.Writer) int {
	if len(args) == 0 {
		fmt.Fprint(stderr, usage)
		return 2
	}
	switch args[0] {
	case "clip":
		return runClipCmd(args[1:], stdout, stderr)
	case "deps":
		if len(args) == 2 && args[1] == "install" {
			return runDepsInstall(baseDir(), stdout, stderr)
		}
		fmt.Fprint(stderr, usage)
		return 2
	case "help", "-h", "--help":
		fs := newClipFlags(&clipOpts{}, new(string), new(string), new(string))
		fs.SetOutput(stdout)
		fs.Usage()
		return 0
	}
	fmt.Fprintf(stderr, "error: unknown command %q\n\n%s", args[0], usage)
	return 2
}

func main() { os.Exit(realMain(os.Args[1:], os.Stdout, os.Stderr)) }

func checkSignature(t *testing.T, path, format string) {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil || len(b) < 12 {
		t.Fatalf("%s: unreadable or too small (%v, %d bytes)", path, err, len(b))
	}
	switch format {
	case "gif":
		if !bytes.HasPrefix(b, []byte("GIF89a")) {
			t.Errorf("%s: not a GIF89a", path)
		}
	case "webp":
		if string(b[0:4]) != "RIFF" || string(b[8:12]) != "WEBP" {
			t.Errorf("%s: not a WebP", path)
		}
	case "mp4", "mov":
		if string(b[4:8]) != "ftyp" {
			t.Errorf("%s: no ftyp box", path)
		}
	}
}

// Installs the pinned tools into a temp dir, checks the encoders the tool
// relies on exist, then encodes real sources to every format.
func TestRealToolsEncode(t *testing.T) {
	d, err := loadDeps("deps.json")
	if err != nil {
		t.Fatal(err)
	}
	binDir := t.TempDir()
	for _, name := range []string{"yt-dlp", "ffmpeg"} {
		if err := installTool(name, d[name], binDir, os.Stdout); err != nil {
			t.Fatalf("install %s: %v", name, err)
		}
	}
	ffmpeg := filepath.Join(binDir, exeName("ffmpeg"))

	encoders, err := exec.Command(ffmpeg, "-hide_banner", "-encoders").Output()
	if err != nil {
		t.Fatal(err)
	}
	for _, enc := range []string{"libx264", "libwebp_anim"} {
		if !bytes.Contains(encoders, []byte(enc)) {
			t.Errorf("pinned ffmpeg lacks %s", enc)
		}
	}

	work := t.TempDir()
	mk := func(args ...string) {
		t.Helper()
		full := append([]string{"-hide_banner", "-loglevel", "error", "-y"}, args...)
		if out, err := exec.Command(ffmpeg, full...).CombinedOutput(); err != nil {
			t.Fatalf("ffmpeg %v: %v\n%s", args, err, out)
		}
	}
	withAudio := filepath.Join(work, "audio.mp4")
	mk("-f", "lavfi", "-i", "testsrc=duration=3:size=320x240:rate=15", "-f", "lavfi", "-i", "sine=duration=3",
		"-shortest", "-c:v", "libx264", "-pix_fmt", "yuv420p", "-c:a", "aac", withAudio)
	// Odd dimensions and no audio track: the two cases mp4/mov must survive.
	oddNoAudio := filepath.Join(work, "odd.mkv")
	mk("-f", "lavfi", "-i", "testsrc=duration=3:size=321x241:rate=15", "-c:v", "ffv1", "-pix_fmt", "yuv444p", oddNoAudio)

	enc := ffmpegEncode(ffmpeg)
	for _, src := range []string{withAudio, oddNoAudio} {
		for _, format := range allFormats {
			src, format := src, format
			t.Run(fmt.Sprintf("%s-%s", filepath.Base(src), format), func(t *testing.T) {
				out := filepath.Join(work, filepath.Base(src)+"."+format)
				if err := enc(context.Background(), src, out, encodeOpts{format, presets["medium"]}); err != nil {
					t.Fatal(err)
				}
				checkSignature(t, out, format)
			})
		}
	}
}
